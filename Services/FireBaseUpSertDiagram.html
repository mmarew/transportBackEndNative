// Pseudocode for the upsertDeviceToken function
const upsertDeviceToken = async ({ userUniqueId, roleId, token, platform, appVersion, locale }) => {
    // 1. Check if token exists for any user
    const existingToken = await db.query('SELECT * FROM DeviceTokens WHERE token = ?', [token]);
    
    if (existingToken) {
        // 2. Token exists - check if same user/role
        if (existingToken.userUniqueId === userUniqueId && existingToken.roleId === roleId) {
            // 3. Update existing token metadata
            await db.query(
                'UPDATE DeviceTokens SET platform=?, appVersion=?, locale=?, lastSeenAt=? WHERE token=?',
                [platform, appVersion, locale, new Date(), token]
            );
            return { action: 'updated' };
        } else {
            // 4. Reassign token to new user/role
            await db.query(
                'UPDATE DeviceTokens SET userUniqueId=?, roleId=?, platform=?, appVersion=?, locale=?, lastSeenAt=? WHERE token=?',
                [userUniqueId, roleId, platform, appVersion, locale, new Date(), token]
            );
            return { action: 'reassigned' };
        }
    } else {
        // 5. Token doesn't exist - check if user has other tokens
        const userTokens = await db.query(
            'SELECT * FROM DeviceTokens WHERE userUniqueId = ? AND roleId = ?',
            [userUniqueId, roleId]
        );
        
        // 6. Insert new token (user can have multiple devices)
        await db.query(
            'INSERT INTO DeviceTokens (deviceTokenUniqueId, userUniqueId, token, platform, appVersion, locale, lastSeenAt, roleId) VALUES (?, ?, ?, ?, ?, ?, ?, ?)',
            [uuidv4(), userUniqueId, token, platform, appVersion, locale, new Date(), roleId]
        );
        
        return { action: userTokens.length > 0 ? 'created_additional' : 'created_first' };
    }
};
